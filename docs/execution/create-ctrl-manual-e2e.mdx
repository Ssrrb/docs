La conexión a PostgreSQL pertenece al Worker; el miembro no recibe sus credenciales.
Esta entrega utiliza la base del servicio con aislamiento por empresa; no provisiona una base separada para cada organización.

```bash
Promise.all([
  "https://clerk.ctrlmas.tech/.well-known/openid-configuration",
  process.env.CTRL_API_URL + "/healthz"
].map(async url => {
  const response = await fetch(url, { signal: AbortSignal.timeout(10000) });
  if (!response.ok) throw new Error("HTTP " + response.status + " en " + url);
  const data = await response.json();
  if (url.endsWith("/healthz") && data.ok !== true) throw new Error("API sin healthz válido");
  if (url.endsWith("/openid-configuration") && data.issuer !== "https://clerk.ctrlmas.tech") throw new Error("Issuer inesperado");

})).catch(error => { console.error(error.message); process.exitCode = 1; });
'
```
